@echo off
title Atualizar G7Rastreadores - GitHub
color 0A

echo ==========================================
echo       ATUALIZANDO G7RASTREADORES
echo ==========================================
echo.

cd /d "%~dp0"

echo [1/4] Adicionando arquivos...
git add .

echo.
echo [2/4] Criando commit...
git commit -m "Atualizacao do site"

echo.
echo [3/4] Buscando atualizacoes do GitHub...
git pull --rebase origin main

if errorlevel 1 (
    echo.
    echo ==========================================
    echo ERRO NO GIT PULL!
    echo Verifique se existe algum conflito.
    echo ==========================================
    pause
    exit /b 1
)

echo.
echo [4/4] Enviando para o GitHub...
git push

if errorlevel 1 (
    echo.
    echo ==========================================
    echo ERRO AO ENVIAR PARA O GITHUB!
    echo ==========================================
    pause
    exit /b 1
)

echo.
echo ==========================================
echo     SITE ATUALIZADO COM SUCESSO!
echo ==========================================
echo.
pause
